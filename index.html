Python Indicator Pipeline & Statistical Analytics

# automated_meal_pipeline.py (Pandas + LogAlto API + Excel Export) Status: Production Ready 
import pandas as pd
import numpy as np
import requests

def fetch_and_clean_indicators(api_url, token):
    headers = {'Authorization': f'Bearer {token}'}
    response = requests.get(api_url, headers=headers)
    df = pd.DataFrame(response.json()['data'])
    
    # 1. Data Cleaning & Standardization
    df['submission_date'] = pd.to_datetime(df['submission_date'])
    df.drop_duplicates(subset=['beneficiary_id', 'project_code'], keep='last', inplace=True)
    df['assistance_value'] = df['assistance_value'].fillna(df.groupby('project_code')['assistance_value'].transform('median'))
    
    # 2. Statistical Z-Score Anomaly Detection
    df['z_score'] = np.abs((df['assistance_value'] - df['assistance_value'].mean()) / df['assistance_value'].std())
    df['is_outlier'] = df['z_score'] > 3.0
    
    # 3. Export Clean Master File for Power BI
    with pd.ExcelWriter('Master_MEAL_Indicators_2026.xlsx', engine='openpyxl') as writer:
        df[df['is_outlier'] == False].to_excel(writer, sheet_name='Clean_Data', index=False)
        df[df['is_outlier'] == True].to_excel(writer, sheet_name='QA_Flagged_Records', index=False)
    return df.describe()
